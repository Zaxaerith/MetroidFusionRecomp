cmake_minimum_required(VERSION 3.20)
project(MetroidFusionRecomp C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/gbarecomp" CACHE PATH "Pinned project framework")
set(GBARECOMP_GENERATED_BIOS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/bios" CACHE PATH "Local BIOS translation")
set(GBARECOMP_TOMLPP_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/tomlplusplus" CACHE PATH "Pinned local toml++")
set(RECOMP_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/recomp-ui" CACHE PATH "Pinned launcher UI")
set(GBARECOMP_SELFHEAL_RECOMPILE_DEFAULT OFF CACHE BOOL "Preview uses interpreter fallback without a compiler" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated/cart/dispatch_table.cpp" OR
   NOT EXISTS "${GBARECOMP_GENERATED_BIOS_DIR}/bios_recompiled.cpp")
    message(FATAL_ERROR "Run tools/regen.ps1 first")
endif()
add_subdirectory("${GBARECOMP_ROOT}" framework EXCLUDE_FROM_ALL)
# Do not bake this developer checkout into portable executables. Local
# opt-in healing can still find headers relative to the project root.
get_target_property(RUNTIME_DEFINES gbarecomp_runtime COMPILE_DEFINITIONS)
list(FILTER RUNTIME_DEFINES EXCLUDE REGEX "^GBARECOMP_SRC_DIR=")
set_property(TARGET gbarecomp_runtime PROPERTY COMPILE_DEFINITIONS ${RUNTIME_DEFINES} GBARECOMP_SRC_DIR="gbarecomp")
file(GLOB CART_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/cart/*.cpp")
add_executable(MetroidFusionRecomp src/main.cpp ${CART_SOURCES})
target_link_gbarecomp_runtime_stack(MetroidFusionRecomp)
set(SHARDS ${CART_SOURCES})
list(FILTER SHARDS INCLUDE REGEX "/recompiled_[0-9]+\\.cpp$")
set_source_files_properties(${SHARDS} PROPERTIES COMPILE_OPTIONS "-O1;-g0")

# Keep UI flags and Dear ImGui out of the large generated cartridge shards.
include("${RECOMP_UI_ROOT}/recomp_ui.cmake")
add_library(game_launcher_ui STATIC src/game_launcher_boot.cpp)
target_include_directories(game_launcher_ui PRIVATE "${GBARECOMP_ROOT}/src/runtime" "${GBARECOMP_ROOT}/src/gba")
recomp_target_launcher_ui(game_launcher_ui CONSOLE gba)
target_include_directories(game_launcher_ui PRIVATE ${SDL2_INCLUDE_DIR})
target_link_libraries(game_launcher_ui PUBLIC ${SDL2_LIBRARY})
target_link_libraries(MetroidFusionRecomp PRIVATE game_launcher_ui)
recomp_stage_launcher_assets(MetroidFusionRecomp CONSOLE gba)
add_executable(launcher_contract tests/launcher_contract.cpp)
target_compile_definitions(launcher_contract PRIVATE RECOMP_LAUNCHER)
target_include_directories(launcher_contract PRIVATE "${GBARECOMP_ROOT}/src/runtime" "${RECOMP_UI_ROOT}/src" "${RECOMP_UI_ROOT}/src/common")
target_link_libraries(launcher_contract PRIVATE game_launcher_ui gbarecomp_gba)
gbarecomp_stage_mingw_runtime(MetroidFusionRecomp)
add_custom_command(TARGET MetroidFusionRecomp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL2_DLL}" "$<TARGET_FILE_DIR:MetroidFusionRecomp>/SDL2.dll")
